    """初始化结构化日志。"""
    structlog.configure(
        processors=[
            structlog.contextvars.merge_contextvars,
            structlog.processors.add_log_level,
            structlog.processors.TimeStamper(fmt="iso"),
            structlog.processors.StackInfoRenderer(),

def get_engine(database_url: str, echo: bool = False) -> AsyncEngine:
    """获取或创建异步引擎。"""
    if database_url not in _engines:
        _engines[database_url] = create_async_engine(
            database_url,
            echo=echo,
            connect_args={
                "timeout": 30,
                "check_same_thread": False,
            },
            poolclass=StaticPool,  # SQLite 单连接池
        )
        # WAL 模式 + busy_timeout
        @event.listens_for(engine.sync_engine, "connect")
        def _set_sqlite_pragma(dbapi_connection, connection_record):
            cursor = dbapi_connection.cursor()
            cursor.execute("PRAGMA journal_mode=WAL")
            cursor.execute("PRAGMA busy_timeout=5000")
            cursor.execute("PRAGMA foreign_keys=ON")
            cursor.close()
    return _engines[database_url]

### 2.3 M2 LLM 网关服务

#### 2.3.1 统一网关（gateway.py）

```python
class LLMGateway:
    """LLM 统一网关。"""

    async def chat(
        self,
        messages: list[dict],
        scenario: str = "default",
        task_id: str = "",
        temperature: float = 0.7,
        max_tokens: int = 2048,
    ) -> LLMResponse:
        """统一聊天接口。

        流程：预算检查 → 选择Provider → 指数退避重试(3次) → 计费 → 持久化用量
        """
        # 1. 预算检查（P1-9）
        daily_cost = await self._get_daily_cost()
        if daily_cost >= self.config.budget_limit:
            raise LLMBudgetExceededException(
                f"今日费用 {daily_cost:.2f}元 已达上限 {self.config.budget_limit}元"
            )
        if daily_cost >= self.config.budget_warning:
            logger.warning("llm_budget_warning", cost=daily_cost)
            # 强制使用 Mock 模式
            provider = MockProvider()
        else:
            provider = self._get_provider()

        # 2. 指数退避重试
        for attempt in range(self.config.llm_max_retries):
            try:
                response = await provider.chat(messages, temperature, max_tokens)
                break
            except Exception as e:
                if attempt == self.config.llm_max_retries - 1:
                    raise
                wait_time = 2 ** attempt
                await asyncio.sleep(wait_time)

        # 3. 计费 + 持久化
        cost = self._calculate_cost(response.model, response.usage)
        await self._persist_usage(response, cost, task_id)
        return response
```

**预算控制（P1-9）**：
- ¥180 预警：新任务强制使用 Mock 模式
- ¥200 拒绝：返回 50003 错误，拒绝新任务

#### 2.7.2 Judge 评分器（judge.py）

LLM-as-Judge，4 维度评分（0-10）：
- accuracy（准确性）：无事实错误/幻觉
- relevance（相关性）：与问题直接相关
- completeness（完整性）：覆盖所有方面
- format（格式）：清晰/结构合理

#### 2.7.3 评分计算（scorer.py）

- 加权综合分（默认权重：accuracy 0.30 / relevance 0.25 / completeness 0.25 / format 0.20）
- 通过阈值：≥6.0
- 统计指标：count/mean/median/std/min/max/pass_rate
- 分组统计：按类别/按难度

#### 2.7.4 完成检测（T6-6）

`completed_cases == total_cases` 时自动：
1. 标记运行完成
2. 计算平均分
3. 生成报告（调用 M7）

#### 2.7.5 一致性校验（T6-7）

校验：
- 评分记录数 == completed_cases
- 通过数（overall≥6）== passed_cases

## 三、接口定义

### 3.1 任务接口

#### 3.1.1 创建任务

**POST** `/api/v1/tasks`

**请求参数**：

| 字段 | 类型 | 必填 | 说明 |
|---|---|---|---|
| input | string | 是 | 任务描述（1-10000字符） |
| scenario | string | 否 | 任务类型：orchestration/eval（默认orchestration） |
| callback_url | string | 否 | 完成回调 URL |
| max_rounds | integer | 否 | 最大轮次（1-20，默认5） |

**返回结构**：

```json
{
  "code": 0,
  "message": "success",
  "data": {
    "id": "uuid",
    "scenario": "orchestration",
    "input": "任务描述",
    "status": "pending",
    "result": null,
    "error_message": null,
    "trace_id": null,
    "callback_url": null,
    "created_at": "2026-10-06T00:00:00",
    "started_at": null,
    "completed_at": null
  },
  "request_id": "trace-id",
  "timestamp": "2026-10-06T00:00:00Z"
}
```

**错误码**：10002（参数校验失败）

#### 3.1.2 任务列表

**GET** `/api/v1/tasks?status=&page=&page_size=`

**查询参数**：

| 字段 | 类型 | 必填 | 说明 |
|---|---|---|---|
| status | string | 否 | 状态筛选：pending/running/completed/failed/cancelled |
| page | integer | 否 | 页码（默认1） |
| page_size | integer | 否 | 每页数量（1-100，默认20） |

#### 3.1.3 取消任务

**POST** `/api/v1/tasks/{task_id}/cancel`

**错误码**：20001（任务不存在）、20002（状态不允许取消）

### 3.2 评测接口

#### 3.2.1 创建评测运行

**POST** `/api/v1/eval/runs`

**请求参数**：

| 字段 | 类型 | 必填 | 说明 |
|---|---|---|---|
| name | string | 是 | 运行名称（1-200字符） |
| model_settings | object | 否 | 模型配置（默认{}） |
| case_ids | array | 否 | 指定用例ID列表 |
| category | string | 否 | 按类别筛选 |

#### 3.2.2 下载报告

**GET** `/api/v1/eval/runs/{run_id}/report/download?format=html`

**查询参数**：format（html/json，默认html）

**错误码**：30003（报告不存在）

## 六、部署与启动

### 6.1 环境配置

```bash
# 1. 克隆项目
git clone <repo-url>
cd multi-agent-orchestration-eval-platform

# 2. 创建虚拟环境
python -m venv .venv
.venv\Scripts\activate  # Windows
# source .venv/bin/activate  # Linux/Mac

# 3. 安装依赖
pip install -r requirements.txt

# 4. 配置环境变量
cp .env.example .env
# 编辑 .env：
# APP_LLM_PROVIDER=mock          # 开发用mock，生产用deepseek
# APP_DEEPSEEK_API_KEY=sk-xxx    # DeepSeek API Key
# APP_CELERY_ENABLED=false       # 开发用同步，生产用true
```

### 6.2 启动命令

```bash
# 开发模式（Mock LLM，无需API Key）
$env:APP_LLM_PROVIDER="mock"
uvicorn app.main:app --host 0.0.0.0 --port 8000 --reload

# 生产模式（真实LLM + Celery）
$env:APP_LLM_PROVIDER="deepseek"
$env:APP_DEEPSEEK_API_KEY="sk-xxx"
$env:APP_CELERY_ENABLED="true"

# 启动 API 服务
uvicorn app.main:app --host 0.0.0.0 --port 8000 --workers 2

# 启动 Celery Worker（另一个终端）
celery -A app.workers.celery_app.celery_app worker --loglevel=info --concurrency=2

# 启动 Celery Beat（定时任务，可选）
celery -A app.workers.celery_app.celery_app beat --loglevel=info
```

### 6.3 Docker 部署

```bash
# 一键启动（API + Redis + Worker）
docker-compose up -d

# 包含 Ollama 本地模型
docker-compose --profile ollama up -d

# 查看日志
docker-compose logs -f api

# 停止
docker-compose down
```
