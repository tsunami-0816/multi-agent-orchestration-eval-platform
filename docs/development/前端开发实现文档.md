### 2.2 统一API请求封装

```javascript
/**
 * 统一API请求封装
 * 自动处理统一响应格式 {code, message, data}
 * code !== 0 时抛出异常
 */
async function apiRequest(url, options = {}) {
    const defaultOptions = { headers: { 'Content-Type': 'application/json' } };
    const opts = { ...defaultOptions, ...options };
    try {
        const resp = await fetch(url, opts);
        const data = await resp.json();
        if (data.code !== 0) {
            throw new Error(data.message || `请求失败 (code: ${data.code})`);
        }
        return data.data;  // 直接返回data字段，调用方无需解构
    } catch (e) {
        console.error(`API请求失败: ${url}`, e);
        throw e;  // 向上抛出，由调用方处理UI反馈
    }
}
```

**交互逻辑**：
- 所有接口调用统一走 `apiRequest()`，自动处理 `code/message/data` 格式
- 成功时直接返回 `data` 字段，简化调用方代码
- 失败时抛出包含 `message` 的Error，调用方catch后展示Toast
- 自动注入 `Content-Type: application/json` 头

/**
 * 创建任务表单校验
 * 校验规则：
 * - input: 必填，1-10000字符
 * - max_rounds: 1-20整数
 * - callback_url: 可选，URL格式（浏览器原生校验）

function validateTaskForm() {
    let valid = true;
    const input = document.getElementById('task-input').value.trim();
    const rounds = parseInt(document.getElementById('task-rounds').value);

    } else {
        fgInput.classList.remove('has-error');

        fgRounds.classList.add('has-error');
        valid = false;
    } else {
        fgRounds.classList.remove('has-error');
    }
    return valid;

#### 2.3.2 创建任务与状态轮询

```javascript
/**
 * 创建任务
 * 流程：表单校验 → 禁用按钮 → POST /tasks → 成功提示 → 重置表单 → 刷新列表 → 启动轮询
 */
async function createTask() {
    if (!validateTaskForm()) { showToast('请检查表单输入', 'warning'); return; }

    const btn = document.getElementById('btn-create-task');
    btn.disabled = true;  // 防重复提交

    const payload = {
        input: document.getElementById('task-input').value.trim(),
        scenario: document.getElementById('task-scenario').value,
        max_rounds: parseInt(document.getElementById('task-rounds').value),
    };
    const callback = document.getElementById('task-callback').value.trim();
    if (callback) payload.callback_url = callback;  // 可选字段，空值不传

    try {
        const task = await apiRequest(`${API_BASE}/tasks`, {
            method: 'POST',
            body: JSON.stringify(payload)
        });
        showToast(`任务创建成功: ${task.id.substring(0, 8)}...`, 'success');
        resetTaskForm();
        loadTasks();
        startTaskPolling(task.id);  // 自动轮询新任务状态
    } catch (e) {
        showToast(`创建失败: ${e.message}`, 'error');
    } finally {
        btn.disabled = false;  // 无论成功失败都恢复按钮
    }
}

/**
 * 单任务状态轮询
 * 每2秒查询一次，最多30次（约1分钟），任务终态后自动停止
 */
function startTaskPolling(taskId) {
    let attempts = 0;
    const maxAttempts = 30;
    const interval = setInterval(async () => {
        attempts++;
        if (attempts > maxAttempts) { clearInterval(interval); return; }
        try {
            const task = await apiRequest(`${API_BASE}/tasks/${taskId}`);
            // 到达终态（completed/failed/cancelled）停止轮询并通知
            if (['completed', 'failed', 'cancelled'].includes(task.status)) {
                clearInterval(interval);
                showToast(`任务${task.status === 'completed' ? '已完成' : '执行结束'}`,
                    task.status === 'completed' ? 'success' : 'warning');
                loadTasks();  // 刷新列表显示最新状态
            }
        } catch (e) {
            if (attempts > 3) clearInterval(interval);  // 连续失败3次停止
        }
    }, 2000);
}
```

### 2.4 执行链路可视化页面

#### 2.4.1 按轮次分组算法

```javascript
/**
 * 将消息按轮次分组
 * 规则：每次遇到 planner 节点视为新一轮开始（第一轮除外）
 * 因为 Agent 流水线为 Planner → Executor → Reviewer → Aggregator，
 * 每轮从 Planner 开始，通过 node 字段变化推断轮次边界
 */
function groupMessagesByRound(messages) {
    const rounds = [];
    let currentRound = [];
    let lastNode = null;
    let plannerCount = 0;

    messages.forEach(m => {
        if (m.node === 'planner' && lastNode !== 'planner') {
            plannerCount++;
            if (plannerCount > 1 && currentRound.length > 0) {
                rounds.push(currentRound);  // 上一轮结束
                currentRound = [];           // 开始新一轮
            }
        }
        currentRound.push(m);
        lastNode = m.node;
    });
    if (currentRound.length > 0) rounds.push(currentRound);
    return rounds;
}
```

#### 2.5.1 Canvas四维度柱状图

```javascript
/**
 * 使用原生Canvas绘制四维度评分对比柱状图
 * 不依赖Chart.js等图表库，零外部依赖
 * 包含：坐标轴、Y轴刻度、渐变柱子、数值标签、X轴标签
 */
function drawDimensionChart(dimensions) {
    const canvas = document.getElementById('chart-dimensions');
    const ctx = canvas.getContext('2d');
    // HiDPI适配：按devicePixelRatio缩放，保证高分屏清晰
    const dpr = window.devicePixelRatio || 1;
    const rect = canvas.getBoundingClientRect();
    canvas.width = rect.width * dpr;
    canvas.height = 220 * dpr;
    ctx.scale(dpr, dpr);

    const labels = ['准确性', '相关性', '完整性', '格式'];
    const values = [dimensions.accuracy, dimensions.relevance, dimensions.completeness, dimensions.format];
    const colors = ['#667eea', '#10b981', '#f59e0b', '#8b5cf6'];

    // 绘制坐标轴和网格线
    ctx.strokeStyle = '#e5e7eb';
    // Y轴刻度（0-10，每2一格）
    for (let i = 0; i <= 5; i++) {
        const val = i * 2;
        const y = padT + chartH - (val / 10) * chartH;
        ctx.fillText(val.toString(), padL - 6, y + 4);
        // 水平网格线
        ctx.beginPath(); ctx.moveTo(padL, y); ctx.lineTo(W - padR, y); ctx.stroke();
    }

    // 绘制渐变柱子
    values.forEach((v, i) => {
        const x = padL + gap * i + (gap - barW) / 2;
        const h = (v / 10) * chartH;
        const y = padT + chartH - h;
        // 每根柱子独立渐变色（上深下浅）
        const grad = ctx.createLinearGradient(x, y, x, y + h);
        grad.addColorStop(0, colors[i]);
        grad.addColorStop(1, colors[i] + '88');
        ctx.fillStyle = grad;
        ctx.beginPath();
        ctx.roundRect(x, y, barW, h, [4, 4, 0, 0]);  // 圆角矩形
        ctx.fill();
        // 柱子顶部数值标签
        ctx.fillStyle = '#374151';
        ctx.font = 'bold 12px sans-serif';
        ctx.textAlign = 'center';
        ctx.fillText(v.toFixed(1), x + barW / 2, y - 6);
    });
}
```

#### 2.5.2 报告统计计算

```javascript
/**
 * 计算报告统计数据
 * 包含：平均分、通过率、中位数、四维度平均分、分数分布（5段）
 * 数据来源：优先使用后端summary，缺失时从scores数组现算
 */
function calculateReportStats(run, scores, summary) {
    const overalls = scores.map(s => s.overall).filter(v => v != null);
    const passed = overalls.filter(v => v >= 6).length;  // 通过阈值≥6.0
    const total = scores.length || run.total_cases || 0;

    // 四维度平均分
    const dimensions = {
        accuracy: avg(scores.map(s => s.accuracy)),
        relevance: avg(scores.map(s => s.relevance)),
        completeness: avg(scores.map(s => s.completeness)),
        format: avg(scores.map(s => s.format))
    };

    // 五段分数分布：0-2 / 2-4 / 4-6 / 6-8 / 8-10
    const distribution = [0, 0, 0, 0, 0];
    overalls.forEach(v => {
        if (v < 2) distribution[0]++;
        else if (v < 4) distribution[1]++;
        else if (v < 6) distribution[2]++;
        else if (v < 8) distribution[3]++;
        else distribution[4]++;
    });

    return {
        avgScore: summary.mean != null ? summary.mean.toFixed(2) : avg(overalls).toFixed(2),
        passRate: total ? passed / total : 0,
        total, passed,
        median: summary.median != null ? summary.median.toFixed(2) : median(overalls).toFixed(2),
        dimensions,
        distribution
    };
}
```

**接口**：`GET /api/v1/tasks?status=&page=&page_size=`

**请求参数**：

| 参数 | 来源 | 默认值 | 说明 |
|---|---|---|---|
| status | 下拉筛选 | 空（全部） | pending/running/completed/failed/cancelled |
| page | APP_STATE.tasks.page | 1 | 页码 |
| page_size | 下拉选择 | 20 | 10/20/50 |

**数据渲染逻辑**：
- 渲染表格：ID（截断8位+title完整）、描述（截断60字符）、场景、状态徽章、创建时间、耗时、操作按钮
- 操作按钮按状态动态显示：running/pending显示"取消"按钮，全部显示"详情"和"轨迹"
- 调用 `renderPagination()` 渲染分页控件，显示总条数和当前页

#### 3.2.3 取消任务

**接口**：`POST /api/v1/tasks/{id}/cancel`

**交互逻辑**：
- 点击"取消"按钮后 `confirm()` 二次确认
- 成功后Toast提示并刷新列表
- 失败时Toast展示错误信息（如任务不存在20001、状态不允许20002）

### 3.3 轨迹接口对接详情

**接口**：`GET /api/v1/tasks/{id}/messages?limit=200`

**并行请求**：同时请求消息列表和任务详情（`Promise.all`）

**数据渲染逻辑**：
1. **概览统计**：任务状态、消息总数、参与角色数、总耗时
2. **Checkpoint导航**：提取所有唯一node值作为检查点，点击可滚动定位
3. **消息时间线**：
   - 调用 `groupMessagesByRound()` 按轮次分组
   - 每轮可独立展开/折叠，支持"全部展开/全部折叠"
   - 每条消息按角色配色（planner蓝/executor绿/reviewer橙/aggregator紫）
   - 长内容（>300字符）默认折叠，点击"展开全文"查看
   - 工具调用作为子卡片嵌入消息下方，可展开查看参数/结果/错误

### 3.4 评测接口对接详情

#### 3.4.1 创建评测运行

**接口**：`POST /api/v1/eval/runs`

**请求参数**：

| 字段 | 类型 | 必填 | 说明 |
|---|---|---|---|
| name | string | 是 | 运行名称，1-200字符 |
| model_config | object | 否 | `{model: "deepseek-chat"}` |
| category | string | 否 | 用例类别筛选 |
| case_ids | array | 否 | 指定用例ID列表（逗号分隔输入转数组） |

#### 3.4.2 评测运行列表

**接口**：`GET /api/v1/eval/runs?page=&page_size=`

**数据渲染逻辑**：
- 进度条：`completed_cases / total_cases * 100%`
- 通过率：`passed_cases / completed_cases * 100%`
- 操作：查看报告、运行中显示"检测"按钮（触发完成检测）

### 3.5 报告接口对接详情

**并行请求**（`Promise.all`）：
1. `GET /api/v1/eval/runs/{id}` — 运行详情
2. `GET /api/v1/eval/runs/{id}/scores?limit=100` — 评分记录（失败时降级为空数组）
3. `GET /api/v1/eval/runs/{id}/report` — 报告summary（失败时降级为null）

**数据渲染逻辑**：
1. **运行信息栏**：名称、ID、模型、创建时间、状态、下载按钮
2. **统计卡片**（6个）：平均分、通过率、用例总数、通过数、未通过、中位数
3. **四维度对比图**：Canvas绘制accuracy/relevance/completeness/format柱状图
4. **分数分布图**：Canvas绘制5段分布柱状图（0-2/2-4/4-6/6-8/8-10）
5. **用例评分明细表**：可排序（综合分/各维度），展示通过/未通过徽章和评审反馈

**下载接口**：
- HTML：`GET /api/v1/eval/runs/{id}/report/download?format=html`
- JSON：`GET /api/v1/eval/runs/{id}/report/download?format=json`
- 通过 `<a href target="_blank">` 直接下载，无需JS处理

## 四、交互实现

### 4.1 表单校验

| 表单 | 字段 | 校验规则 | 触发时机 | 错误展示 |
|---|---|---|---|---|
| 创建任务 | input | 必填，1-10000字符 | 提交时 + 实时字符计数 | 红色边框 + 下方错误文字 |
| 创建任务 | max_rounds | 1-20整数 | 提交时 | 红色边框 + 下方错误文字 |
| 创建任务 | callback_url | URL格式（可选） | 浏览器原生type=url校验 | 浏览器原生提示 |
| 创建评测 | name | 必填，1-200字符 | 提交时 | 红色边框 + 下方错误文字 |

**校验实现要点**：
- 使用 `.form-group.has-error` CSS类控制错误样式（红色边框+错误文字显示）
- 校验通过时移除 `has-error` 类，恢复正常样式
- 字符计数实时更新（`input` 事件监听）
- 提交前统一调用 `validateTaskForm()`，任一字段不通过则阻止提交并Toast警告

### 4.2 状态反馈

#### 4.2.1 Toast通知系统

```javascript
// 4种类型：success(绿) / error(红) / warning(橙) / info(蓝)
showToast('任务创建成功', 'success');     // 成功操作
showToast('创建失败: xxx', 'error');       // 错误操作
showToast('请检查表单输入', 'warning');    // 警告提示
showToast('已开启自动刷新', 'info');       // 信息提示
```

**特性**：
- 右上角固定定位，堆叠展示
- 3秒后自动消失（带淡出+位移动画）
- 每种类型带对应图标（✓/✕/⚠/ℹ）
- 操作结果必有反馈，用户不会"点了没反应"

#### 4.2.2 加载状态

| 场景 | 加载展示 |
|---|---|
| 按钮提交中 | 按钮disabled + 文字旁loading spinner |
| 列表加载中 | 表格区域显示"加载中..." + spinner |
| 轨迹加载中 | 消息区域显示"加载执行链路中..." + spinner |
| 报告加载中 | 报告区域显示"加载报告中..." + spinner |
| 健康检查 | 导航栏状态点变橙色 + "检测中..."文字 |

#### 4.2.3 空状态与错误状态

- **空状态**：居中图标 + 提示文字 + 操作引导（如"输入任务ID后查看"）
- **错误状态**：居中警告图标 + 错误信息 + 可能的原因提示
- 所有列表/详情加载失败都有明确的错误展示，不会白屏

### 4.3 弹窗逻辑

**任务详情弹窗**（`.modal-overlay`）：
- 点击"详情"按钮打开，展示任务完整信息
- 点击右上角×或弹窗外部区域关闭
- 包含"查看执行轨迹"快捷按钮（关闭弹窗+跳转轨迹页）
- 运行中任务显示"取消任务"按钮
- 打开时 `fadeIn + slideUp` 动画

### 4.4 异常处理

| 异常场景 | 处理方式 |
|---|---|
| API请求失败（网络错误） | `apiRequest()` catch后抛出，调用方展示error Toast + 页面错误状态 |
| 后端返回code≠0 | `apiRequest()` 自动检测并抛出 `message`，调用方展示 |
| 任务不存在（20001） | Toast展示"任务不存在"，列表刷新 |
| 状态不允许取消（20002） | Toast展示具体错误信息 |
| 报告接口失败 | `Promise.all` 中使用 `.catch(() => null)` 降级，不影响其他数据渲染 |
| 评分接口失败 | 同上，降级为空数组，报告仍可展示运行基本信息 |
| 轮询连续失败3次 | 自动停止轮询，避免无限请求 |
| 轮询超过30次（1分钟） | 自动停止，避免长时间运行 |
| XSS攻击 | 所有动态内容通过 `escapeHtml()` 转义后再插入DOM |

**降级策略**：报告页采用渐进式渲染，3个并行请求中任一失败不影响其他数据展示，确保页面始终有内容可看。

**前端文件位置**：`static/index.html`，由 FastAPI `StaticFiles` 中间件托管。

**后端静态文件挂载配置**（`app/main.py`）：
```python
from fastapi.staticfiles import StaticFiles
app.mount("/static", StaticFiles(directory="static"), name="static")
```

### 5.3 打包方式

由于采用纯静态单文件架构，**无需构建打包**：

- **开发环境**：直接修改 `static/index.html`，浏览器刷新即可生效
- **生产部署**：`static/index.html` 随后端代码一起部署，FastAPI直接托管
- **Docker部署**：Dockerfile中 `COPY static/ /app/static/` 即可，无需Node构建步骤
- **CDN部署**：可将 `index.html` 单独上传到CDN，修改 `API_BASE` 为后端地址即可

**如需独立部署前端**（脱离后端）：
1. 将 `index.html` 放到任意静态文件服务器（Nginx/Apache/CDN）
2. 修改文件中 `const API_BASE = '/api/v1'` 为后端完整地址（如 `'http://api.example.com/api/v1'`）
3. 后端需配置CORS允许前端域名跨域访问
